  $clean = New-Object System.Text.StringBuilder
  $spans = New-Object System.Collections.ArrayList
  $italic = $false
  $start = 0
  foreach ($ch in $t.ToCharArray()) {
    if ($ch -eq '*') {
      if ($italic) { [void]$spans.Add(@{ S = $start; L = $clean.Length - $start }) ; $italic = $false }
      else { $italic = $true; $start = $clean.Length }
      continue
    }
    [void]$clean.Append($ch)
  }
  $paras[$i].C = $clean.ToString()
  $paras[$i].S = $spans
}
